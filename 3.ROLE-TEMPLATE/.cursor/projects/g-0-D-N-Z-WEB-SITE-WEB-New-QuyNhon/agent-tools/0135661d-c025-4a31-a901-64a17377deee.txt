<script>
    (function () {
        var popover = document.getElementById('admissionPopover');
        var btn = document.getElementById('admissionTriggerBtn');
        var container = document.getElementById('admissionWidgetContainer');

        window.toggleAdmissionPopover = function (event) {
            event.stopPropagation();
            popover.classList.toggle('active');
            btn.classList.toggle('active-btn');
        };

        document.addEventListener('click', function (event) {
            if (!container.contains(event.target)) {
                popover.classList.remove('active');
                btn.classList.remove('active-btn');
            }
        });

        document.addEventListener('keydown', function (event) {
            if (event.key === 'Escape') {
                popover.classList.remove('active');
                btn.classList.remove('active-btn');
            }
        });
    })();
</script>
